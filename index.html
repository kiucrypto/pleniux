const express = require('express'); const axios = require('axios'); const bodyParser = require('body-parser'); const app = express(); app.use(bodyParser.json()); app.use(express.static('public')); // Carpeta donde estará tu HTML // Base de datos en memoria (puedes adaptarla a MongoDB o PostgreSQL en producción) const userBalances = {}; // { "UX123": 5 } const processedTxIds = new Set(); // Evita la reutilización de TxIDs // Tu dirección de billetera real configurada const MY_REAL_WALLET = "bc1qxy2kgdygjrsqtzq2n0yrf2493p83kkfjhx0wlh"; // Endpoint para verificar el pago de forma 100% real en la Blockchain app.post('/api/verify-payment', async (req, res) => { const { nodeID, txHash, expectedAmountUSD, cryptoAsset } = req.body; if (!nodeID || !txHash) { return res.status(400).json({ success: false, message: "Datos incompletos." }); } if (processedTxIds.has(txHash)) { return res.status(400).json({ success: false, message: "Este TxID ya fue utilizado anteriormente." }); } try { // Consulta real a la API pública de BlockCypher para Bitcoin const response = await axios.get(`https://api.blockcypher.com/v1/btc/main/txs/${txHash}`); const txData = response.data; let paymentFound = false; let totalReceivedSatoshis = 0; if (txData && txData.outputs) { txData.outputs.forEach(output => { if (output.addresses && output.addresses.includes(MY_REAL_WALLET)) { paymentFound = true; totalReceivedSatoshis += output.value; } }); } if (!paymentFound) { return res.status(400).json({ success: false, message: "El TxID no corresponde a un envío dirigido a tu billetera real." }); } if (txData.confirmations < 1) { return res.status(400).json({ success: false, message: "La transacción está en la mempool pero aún no tiene confirmaciones en la red. Espera un momento." }); } processedTxIds.add(txHash); if (!userBalances[nodeID]) userBalances[nodeID] = 5; let creditsToAdd = 10; if (expectedAmountUSD === "3.99") creditsToAdd = 30; if (expectedAmountUSD === "5.99") creditsToAdd = 50; if (expectedAmountUSD === "29.99") creditsToAdd = 220; if (expectedAmountUSD === "49.99") creditsToAdd = 530; if (expectedAmountUSD === "79.99") creditsToAdd = 810; if (expectedAmountUSD === "666.00") creditsToAdd = 4050; if (expectedAmountUSD === "1299.00") creditsToAdd = 8050; userBalances[nodeID] += creditsToAdd; return res.json({ success: true, message: `¡Pago verificado en la blockchain con éxito! Se han acreditado ${creditsToAdd} UX a tu nodo.`, newBalance: userBalances[nodeID] }); } catch (error) { console.error("Error al consultar la blockchain:", error.message); return res.status(500).json({ success: false, message: "No se pudo verificar el TxID en este momento. Comprueba que el hash sea correcto." }); } }); app.get('/api/balance/:nodeID', (req, res) => { const { nodeID } = req.params; const balance = userBalances[nodeID] !== undefined ? userBalances[nodeID] : 5; res.json({ success: true, balance }); }); const PORT = process.env.PORT || 3000; app.listen(PORT, () => { console.log(`Servidor real de pleniux.com corriendo en el puerto ${PORT}`); });
